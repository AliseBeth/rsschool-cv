function mergeArrays(arr1, arr2) {
  let newArr = arr1.concat(arr2);
  
  function sorting(a, b) {
    return a - b;
  };
  
  let result = newArr.sort(sorting);
  let final = [];
  for (let i = 0; i < result.length; i++) {
    if (result[i] === result[i + 1]){
      continue;
    } else {
      final = final.concat(result[i]);
    }
  }
    
  return final; 
};
